# Launch work-group sizes for scripts/sass_parity.py, for kernels whose PTX entry carries no
# .reqntid/.maxntid (DPC++ emits one only for reqd_work_group_size). One `<value> <regex>` per
# line, first match on the demangled kernel name wins; `\N` is capture group N of the regex.
# Each row cites the launch that fixes the size; a kernel with a runtime WG has no row.

# src/ops/potrf/kernels/potrf_tiny.cc potrf_tiny_launch: per_wg * N == kTinyWgSize
64  ^batchlas::PotrfTinyKernel<
# src/ops/posv/kernels/posv_tiny.cc posv_tiny_launch_b: sycl::range<1>(kTinyWg)
64  ^batchlas::PosvTinyKernel<
# src/ops/syev/sytrd/latrd_lower_panel.cc: sycl::range<1>(wg), wg = the WG template argument
\1  ::LatrdLowerPanelKernel(?:Legacy|Grid)?<[^,]+, (\d+), (?:true|false)>$
# src/ops/sygst/kernels/sygst_diag.cc launch_diag: wg = (ipw * KB + 31) / 32 * 32,
# ipw = diag_items_per_group<T>(KB) (sygst_native.hh: kDiagWg 128, kDiagSlmCap 32768 bytes).
# KB = 0 (run-time order) has no row.
96  ::SygstDiagKernel<float, (?:true|false), 32>$
32  ::SygstDiagKernel<.+, (?:true|false), 32>$
64  ::SygstDiagKernel<[^,]*Cx<double>, (?:true|false), 16>$
128 ::SygstDiagKernel<.+, (?:true|false), (?:16|8)>$
